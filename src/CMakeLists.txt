cmake_minimum_required(VERSION 3.25)
project(symbiosic VERSION 0.1.1 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(CYGWIN OR MSYS)
 set(DESKTOP_DEFAULT OFF)
else()
 set(DESKTOP_DEFAULT ON)
endif()
option(SYMBIOSIC_DESKTOP "Embed the OS webview (Windows MSYS uses the native packaging host)" ${DESKTOP_DEFAULT})
include(FetchContent)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(UV REQUIRED IMPORTED_TARGET libuv)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLHTTP_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLHTTP_BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
FetchContent_Declare(yyjson URL https://github.com/ibireme/yyjson/archive/refs/tags/0.13.0.tar.gz URL_HASH SHA256=34e0f62a2bc11ab20d601e8ca1cc2b2079503aa45119a19133d89d19b94a0fae DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_Declare(llhttp URL https://github.com/nodejs/llhttp/archive/refs/tags/release/v9.4.3.tar.gz URL_HASH SHA256=1eb813c7437b31a87496a1cd3ed79f00746720f5e7e29c79b42c02cb69f36c39 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(yyjson llhttp)
set(TIKTOKEN_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps" CACHE PATH "Tokenizer header and library")
add_library(tiktoken STATIC IMPORTED)
set_target_properties(tiktoken PROPERTIES IMPORTED_LOCATION "${TIKTOKEN_DIR}/libtiktoken_c.a" INTERFACE_INCLUDE_DIRECTORIES "${TIKTOKEN_DIR}")
set(SOURCES prompts.c directory.c review.c fidelity.c providers.c util.c store.c normalize.c query.c ai.c chat.c main.c http.c)
function(configure_app target)
 target_compile_definitions(${target} PRIVATE _POSIX_C_SOURCE=200809L _DARWIN_C_SOURCE _DEFAULT_SOURCE)
 target_compile_options(${target} PRIVATE -Wall -Wextra)
 target_link_libraries(${target} PRIVATE PkgConfig::CURL PkgConfig::UV yyjson llhttp_static tiktoken m)
 if(APPLE)
  target_link_libraries(${target} PRIVATE iconv)
 elseif(CYGWIN OR MSYS)
  target_link_libraries(${target} PRIVATE advapi32)
 else()
  target_link_libraries(${target} PRIVATE dl pthread)
 endif()
endfunction()
add_executable(symbiosic ${SOURCES})
configure_app(symbiosic)
if(SYMBIOSIC_DESKTOP)
 if(CYGWIN OR MSYS OR WIN32)
  message(FATAL_ERROR "Build the Windows MSYS backend with SYMBIOSIC_DESKTOP=OFF; scripts/package.py builds its native WebView2 host.")
 endif()
 enable_language(CXX)
 include(cmake/Webview.cmake)
 target_sources(symbiosic PRIVATE desktop.c desktop_window.c)
 target_compile_definitions(symbiosic PRIVATE SYMBIOSIC_DESKTOP)
 target_link_libraries(symbiosic PRIVATE webview::core_static)
 if(APPLE)
  enable_language(OBJC)
  target_sources(symbiosic PRIVATE desktop_macos.m)
  set_source_files_properties(desktop_macos.m PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  target_link_libraries(symbiosic PRIVATE "-framework Cocoa")
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(test_symbiosic tests/test.c ${SOURCES})
 target_include_directories(test_symbiosic PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
 target_compile_definitions(test_symbiosic PRIVATE SYMBIOSIC_TEST)
 configure_app(test_symbiosic)
 add_test(NAME core COMMAND test_symbiosic)
 set_tests_properties(core PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
 find_package(Python3 COMPONENTS Interpreter)
 if(Python3_Interpreter_FOUND)
  add_test(NAME lifecycle COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/lifecycle_test.py" "$<TARGET_FILE:symbiosic>")
 endif()
endif()
