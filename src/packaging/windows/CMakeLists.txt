cmake_minimum_required(VERSION 3.25)
project(symbiosic_windows_host LANGUAGES C CXX RC)
set(CMAKE_C_STANDARD 17)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/Webview.cmake")
add_executable(Symbiosic WIN32 host.c ../../desktop_window.c "${SYMBIOSIC_ICON_RESOURCE}")
target_compile_definitions(Symbiosic PRIVATE UNICODE _UNICODE)
target_compile_options(Symbiosic PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall> $<$<COMPILE_LANGUAGE:C>:-Wextra>)
target_link_libraries(Symbiosic PRIVATE webview::core_static)
if(MINGW)
 target_link_options(Symbiosic PRIVATE -municode -static)
endif()
